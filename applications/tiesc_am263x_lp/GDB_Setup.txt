# Read from an MMR
define Read_MMR
   set $p_mmr = (unsigned int *) $arg0
   # return *$p_mmr
end

# Write to an MMR
define Write_MMR
   set {uint32_t}$arg0 = $arg1
end

# Write to a specific field in an MMR.
define Write_MMR_Field
    #Grab the MMR value
    set $origVal = {uint32_t}$arg0
    #Build a mask of 1s for the field.
    set $mask = (1 << $arg2) - (1 << $arg3)
    #Invert the mask so that the field will be zerod out with the AND operation.
    set $mask = ~($mask)
    #Zero out the field in the register.
    set $origVal = $origVal & $mask
    #Assign the value to that specific field.
    set {uint32_t}$arg0 = $origVal | ($arg1 << $arg3)
end

define Program_SYS_CLK_DIVBY2
   # Enable DIV BY 2 on SYS_CLK (GCM)

   # SYS_CLK_DIV_VAL = 0x111; DIV val=1 means divide by 2

define Program_Core_PLL
   set $val_pll_core_stat = {uint32_t}$MSS_TOP_RCM_PLL_CORE_STATUS
   if (($val_pll_core_stat & 0x00000400) != 0x00000400)
      # N
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_M2NDIV 9 8 0
      # M2
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_M2NDIV 1 7 16
      # M
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_MN2DIV 0x320 12 0
      # N2
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_MN2DIV 0x0 4 16
      # SD divider of the PLL with the value of 0x8 to get the optimum jitter performance
      Write_MMR_Field 0x53200418 8 8 24
      # Idle
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_CLKCTRL 0 1 23
      # Soft Reset
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_CLKCTRL 0 1 0
      # PLL_CORE_TENABLE = 1
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_TENABLE 1 1 0
      # Soft Reset
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_CLKCTRL 1 1 0
      # PLL_CORE_TENABLE = 0
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_TENABLE 0 1 0
      # PLL_CORE_TENABLEDIV = 1
      Write_MMR $MSS_TOP_RCM_PLL_CORE_TENABLEDIV 1
      # PLL_CORE_TENABLEDIV = 0
      Write_MMR $MSS_TOP_RCM_PLL_CORE_TENABLEDIV 0

      # Wait for PLL to start
      set $val_pll_core_stat = {uint32_t}$MSS_TOP_RCM_PLL_CORE_STATUS
      while (($val_pll_core_stat & 0x00000400) != 0x00000400)
         set $val_pll_core_stat = {uint32_t}$MSS_TOP_RCM_PLL_CORE_STATUS
      end

      # CLKOUT0
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER_CLKOUT0 0x4 5 0
      # CLKOUT1
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER_CLKOUT1 0x3 5 0
      # CLKOUT2
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER_CLKOUT2 0x4 5 0
      # PLL_CORE_HSDIV_TENABLEDIV = 1
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER 1 1 2
      # PLL_CORE_HSDIV_TENABLEDIV = 0
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER 0 1 2
      # CLKOUT0
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER_CLKOUT0 0x1 1 8
      # CLKOUT1
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER_CLKOUT1 0x1 1 8
      # CLKOUT1
      Write_MMR_Field $MSS_TOP_RCM_PLL_CORE_HSDIVIDER_CLKOUT2 0x1 1 8
      print "CORE PLL Configuration Complete"
   end
end

# Read from a specific field in an MMR
define Read_MMR_Field
   set $p_mmr = (unsigned int *) $mmr_address
   set $mask = (1 << $width) - 1 << $leftshift # Build a mask of 1s for the field
   set $retval = (*$p_mmr & $mask) >> $leftshift # Bitwise AND the mask with the register value, then shift the field back down to the LSB
   # return $retval
end

    Write_MMR $MSS_CTRL_MAILBOXRAM_MEM_INIT 0x1

        set $val_mem_init_done = ({uint32_t}$MSS_CTRL_MAILBOXRAM_MEM_INIT_DONE)

define AM263x_Cryst_Clock_Loss_Status
   set $val_crystal_clk_loss_stat = {uint32_t}($MSS_TOP_RCM_CLK_LOSS_STATUS)

   if(($val_crystal_clk_loss_stat & 1) == 0x1)
      print "Crystal Clock not present"
   end
   if(($val_crystal_clk_loss_stat & 1) != 0)
      print "Crystal Clock present"
   end
end

define AM263x_SOP_Mode
   set $val_sop_mode = {uint32_t}($MSS_TOP_RCM_SOP_MODE_VALUE)

   printf "SOP MODE = %02X", $val_sop_mode

   if($val_sop_mode == 0)
      print "  QSPI - 4S Functional boot mode"
   end
   if($val_sop_mode == 1)
      print "  UART Functional boot mode"
   end
   if($val_sop_mode == 2)
      print "  QSPI - 1S Functional boot mode"
   end
   if($val_sop_mode == 4)
      print "  QSPI - 4S Fallback UART boot mode"
   end
   if($val_sop_mode == 5)
      print "  QSPI - 1S Fallback UART boot mode"
   end
   if($val_sop_mode == 11)
      print "  Dev boot mode" 
   end
end

define main
   AM263x_Cryst_Clock_Loss_Status
   AM263x_SOP_Mode

   MSS_TOPRCM_unlock
   Configure_Lockstep_mode
   Configure_Plls_R5F_400_SYS_200_Clocks

   Configure_All_Peripheral_Clks

   monitor halt
   monitor wait_halt

   # Disable caches and MMU
   monitor arm mcr 15 0 0 0 2 0x10001
   monitor arm mcr 15 0 1 0 0 0xE50878

   # Reset interrupt controllers
   Write_MMR $MSS_RCM_R5SS0_VIM0_RST_CTRL 0x07070707
   Write_MMR $MSS_RCM_R5SS1_VIM0_RST_CTRL 0x07070707
   Write_MMR $MSS_RCM_R5SS0_VIM1_RST_CTRL 0x07070707
   Write_MMR $MSS_RCM_R5SS1_VIM1_RST_CTRL 0x07070707

   Write_MMR $MSS_RCM_R5SS0_VIM0_RST_CTRL 0x00000000
   Write_MMR $MSS_RCM_R5SS1_VIM0_RST_CTRL 0x00000000
   Write_MMR $MSS_RCM_R5SS0_VIM1_RST_CTRL 0x00000000
   Write_MMR $MSS_RCM_R5SS1_VIM1_RST_CTRL 0x00000000

   # Reset watchdogs
   Write_MMR $MSS_RCM_WDT0_RST_CTRL 0x07070707
   Write_MMR $MSS_RCM_WDT1_RST_CTRL 0x07070707
   Write_MMR $MSS_RCM_WDT2_RST_CTRL 0x07070707
   Write_MMR $MSS_RCM_WDT3_RST_CTRL 0x07070707

   Write_MMR $MSS_RCM_WDT0_RST_CTRL 0x00000000
   Write_MMR $MSS_RCM_WDT1_RST_CTRL 0x00000000
   Write_MMR $MSS_RCM_WDT2_RST_CTRL 0x00000000
   Write_MMR $MSS_RCM_WDT3_RST_CTRL 0x00000000
